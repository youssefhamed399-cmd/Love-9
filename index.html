أُغازلُها وأُسرفُ في هواها

— قصيدة في دودي —

أُغازلُها وأُسرفُ في هواها

كأنَّ اللهَ لم يخلقْ سواها

ولو عيني رأتْ وجهًا جميلًا

فإني دونما وعيٍ أراها

أراها في الصباحِ إذا تنفّسْ

وأبصرُ في النجومِ لها سناها

وفي عينيها بحرٌ من حنينٍ

إذا أبحرتُ فيهِ نسيتُ ميناها

لها وجهٌ إذا أشرقَتْ ملامحُهُ

أضاءَ القلبَ قبلَ أن أراها

وللثغرِ الجميلِ إذا تبسّمْ

ربيعٌ يستحي الوردُ من شذاها

وشَعرُها إذا هبَّتْ نسائمٌ

تمنّتْ كلُّ ريحٍ أن تلامسَها

وصوتُها إذا ناداني همسًا

أضاعَ من الفؤادِ كلَّ جفاها

تمشي فيميلُ الليلُ نحوَ خطاها

وتخجلُ من جمالِها النجومُ إذا رآها

وإن سألوا فؤادي: من تهوى؟

أقولُ بكلِّ فخرٍ: من سواها؟

هيَ الحُسنُ الذي ما زلتُ أروي

حكاياتِ الهوى شوقًا إليها

ولو خُيِّرتُ بينَ الناسِ يومًا

لاخترتُ التي يسكنُ فؤادي هواها 🤍

بحبك يا دودي ♥
